//@version=6
// Satya RS Line vs 21 EMA
// Plots RS Line (stock/benchmark) vs its 9/13/21/200 EMAs and weekly RS EMA9.
// No companion overlay indicator. Separate pane (overlay=false).
// Alert message format: SIGNAL_TYPE|{{ticker}}|description|TF:{{interval}}
indicator("Satya RS Line vs 21 EMA", overlay=false)

/////RS alert
// === Inputs ===
//indexSym = input.symbol("NIFTYMIDSML400", "Index Symbol (NIFTYMIDSML400)")
isNSE = str.contains(syminfo.ticker, "NSE") or str.contains(syminfo.prefix, "NSE")

// Nifty50 constituents (as of 2026-07-17) — TradingView tickers, so "-" and "&"
// become "_" (M&M -> M_M, BAJAJ-AUTO -> BAJAJ_AUTO). Comma-delimited with leading/
// trailing commas so str.contains matches whole tickers only (LT vs ULTRACEMCO).
// Update manually on index rebalance.
nifty50List = ",HDFCBANK,ICICIBANK,RELIANCE,BHARTIARTL,LT,SBIN,INFY,AXISBANK,M_M,KOTAKBANK,BAJFINANCE,ITC,TCS,ETERNAL,SUNPHARMA,TITAN,HINDUNILVR,MARUTI,NTPC,TATASTEEL,BEL,SHRIRAMFIN,HINDALCO,ULTRACEMCO,ADANIPORTS,POWERGRID,HCLTECH,ASIANPAINT,INDIGO,GRASIM,BAJAJ_AUTO,JSWSTEEL,BAJAJFINSV,COALINDIA,EICHERMOT,NESTLEIND,ONGC,TECHM,TRENT,ADANIENT,APOLLOHOSP,SBILIFE,MAXHEALTH,CIPLA,JIOFIN,DRREDDY,TATACONSUM,TMPV,HDFCLIFE,WIPRO,"
isNifty50 = str.contains(nifty50List, "," + syminfo.ticker + ",")

indexSym = isNifty50 ? "NIFTY" : isNSE ? "NIFTYMIDSML400" : "SPX"
emaLen   = input.int(21, "RS EMA Length")

// === Relative Strength Line ===
stockClose  = request.security(syminfo.tickerid, timeframe.period, close)
indexClose  = request.security(indexSym, timeframe.period, close)
rsLine = (stockClose / indexClose) * 1000  // Scaled for visibility

// === 21-day EMA of RS Line ===
rsEma21 = ta.ema(rsLine, emaLen)
rsEma9 = ta.ema(rsLine, 9)
rsEma13 = ta.ema(rsLine, 13)
rsWeeklyEma9 = request.security(syminfo.tickerid, "1W", ta.ema(rsLine, 9))
rsBaseline = ta.ema(rsLine, 50)  // slow EMA as mid line; auto-scales per symbol, unlike a fixed hline
rsEma200 = ta.ema(rsLine, 200)
// === Background: RS EMA9 (daily) above weekly RS EMA9, weekly EMA9 flat/rising ===
weeklyEma9Rising = rsWeeklyEma9 >= rsWeeklyEma9[1]
rsAboveWeeklyEma9 = rsEma9 > rsWeeklyEma9 and weeklyEma9Rising
bgcolor(rsAboveWeeklyEma9 ? color.new(color.teal, 85) : na, title="RS EMA9 > Weekly EMA9 (rising)")

// === Background: RS line crossed latest RS-down high line ===
// lastRsDownHigh = rsLine level of the most recent bar where rsLine was declining
// (mirrors rs_highline_scanner.py's backward scan, done forward via running state)
var float lastRsDownHigh = na
if rsLine < rsLine[1]
    lastRsDownHigh := rsLine
rsDownCross = not na(lastRsDownHigh) and ta.crossover(rsLine, lastRsDownHigh)
//bgcolor(rsDownCross ? color.new(color.orange, 70) : na, title="RS crossed last RS-down high")

// === Plots ===
plot(rsLine, title="RS Line", color=rsLine > rsLine[1] ? color.blue : color.red, linewidth=1, style = plot.style_stepline)
plot(rsEma21, title="RS EMA 21", color=rsEma21 > rsEma21[1]?color.blue : color.red, linewidth=1)
//plot(rsEma13, title="RS EMA 13", color=rsEma13 > rsEma13[1]?color.lime : color.fuchsia, linewidth=1)
plot(rsEma9, title="RS EMA 9", color=rsEma9 > rsEma9[1]?color.aqua : color.purple, linewidth=1)
plot(rsWeeklyEma9, title="RS Week EMA 9", color=rsWeeklyEma9 > rsWeeklyEma9[1]?color.white : color.gray, linewidth=1, style = plot.style_stepline)
//plot(rsBaseline, title="RS Baseline (EMA50)", color=color.gray, linewidth=1, style=plot.style_line)
plot(rsEma200, title="RS Baseline (EMA200)", color=color.yellow, linewidth=1, style=plot.style_line)

// Crossover logic
// bullCross = ta.crossover(rsLine, rsWeeklyEma9)    // 9 crosses above 21
// plotshape(     bullCross,     title="Cross Above rsWeeklyEma9",     style=shape.cross,     location=location.belowbar,     color=color.white,     size=size.tiny)

// === Crossover condition ===
rsCrossUp = ta.crossover(rsLine, rsEma21)// and rsLine > rsEma21
//rsCrossUpNormal = rsCrossUp and not isPPV
//rsCrossUpPPV = rsCrossUp and isPPV

// === Visual marker ===
// plotshape(
//     rsCrossUp ,
//     title     = "Normal RS Line > EMA21",
//     style     = shape.triangleup,
//     location  = location.belowbar,
//     color     = color.fuchsia,
//     size      = size.tiny
// )

// === RS EMA9 / RS EMA21 turn from falling-or-flat to rising ===
rsEma9UptrendStart  = rsEma9 > rsEma9[1] and rsEma9[1] <= rsEma9[2]
rsEma21UptrendStart = rsEma21 > rsEma21[1] and rsEma21[1] <= rsEma21[2]

alertcondition(rsEma9UptrendStart and barstate.isconfirmed, title="RS EMA9 Uptrend Start",
     message="RS_EMA9_UPTREND_START|{{ticker}}|RS EMA9 turned upward|TF:{{interval}}")

alertcondition(rsEma21UptrendStart and barstate.isconfirmed, title="RS EMA21 Uptrend Start",
     message="RS_EMA21_UPTREND_START|{{ticker}}|RS EMA21 turned upward|TF:{{interval}}")
