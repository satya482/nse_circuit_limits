//=== Highest down‑day volume in last N trading days ===
// We only consider bars where `isDown` was true.
// Walk back bar‑by‑bar until we've seen `ppLen` down days or run out of history.
var float maxDownVol = na
var int   seen       = 0
maxDownVol := 0.0
seen       := 0

for i = 1 to 500  // safety cap
    if seen >= ppLen
        break
    if isDown[i]
        maxDownVol := math.max(maxDownVol, volume[i])
        seen += 1

//=== Pocket Pivot condition ===
// Up bar + today’s volume > highest down‑day volume of last N down days
isPPV = isUp and volume > maxDownVol and seen == ppLen

motherDownColor = input.color(color.new(color.black, 0), "Mother Down Bar Color")
extendBars = input.int(10, "Box Extend Bars", minval=1)
lookbackDays = input.int(365, "Lookback Days", minval=30)

var box[] coilBoxes = array.new_box()
var int[] coilStarts = array.new_int()
var int[] coilEnds = array.new_int()
var float[] coilHighs = array.new_float()
var float[] coilLows = array.new_float()

var float coilHigh = na
var float coilLow = na
var int coilStart = na
var int coilEnd = na
var bool coilActive = false

isCoil(len) =>
    mbHigh = high[len]
    mbLow = low[len]
    ok = true
    for i = 0 to len - 1
        ok := ok and high[i] <= mbHigh and low[i] >= mbLow
    ok

cutoffTime = timenow - lookbackDays * 24 * 60 * 60 * 1000
coilFound = bar_index >= minInsideBars and isCoil(minInsideBars)
motherBarIndex = bar_index - minInsideBars
motherBarTime = time[minInsideBars]

if coilFound and motherBarTime >= cutoffTime
    newHigh = high[minInsideBars]
    newLow = low[minInsideBars]
    newStart = motherBarIndex
    newEnd = bar_index + extendBars
    isOverlap = array.size(coilStarts) > 0 and newStart <= array.get(coilEnds, array.size(coilEnds) - 1)
    if isOverlap
        lastBox = array.get(coilBoxes, array.size(coilBoxes) - 1)
        box.delete(lastBox)
        array.pop(coilBoxes)
        array.pop(coilStarts)
        array.pop(coilEnds)
        array.pop(coilHighs)
        array.pop(coilLows)
    newBox = box.new(left=newStart, top=newHigh, right=newEnd, bottom=newLow, bgcolor=coilBoxColor, border_color=coilBorderColor)
    array.push(coilBoxes, newBox)
    array.push(coilStarts, newStart)
    array.push(coilEnds, newEnd)
    array.push(coilHighs, newHigh)
    array.push(coilLows, newLow)
    coilHigh := newHigh
    coilLow := newLow
    coilStart := newStart
    coilEnd := newEnd
    coilActive := true

barcolor(bar_index == motherBarIndex ? (close[minInsideBars] >= open[minInsideBars] ? motherUpColor : motherDownColor) : na)

coilHighCrossover_ = ta.crossover(close, coilHigh)
coilLowCrossunder_ = ta.crossunder(close, coilLow)
coilLowCrossover_ = ta.crossover(close, coilLow)
coilHighCrossover = coilActive and not na(coilHigh) and coilHighCrossover_
coilLowCrossunder = coilActive and not na(coilLow) and coilLowCrossunder_
coilLowCrossover = coilActive and not na(coilLow) and coilLowCrossover_

breakoutUp = coilHighCrossover
breakdownLow = coilLowCrossunder
breakoutLow = coilLowCrossover

showHighLine   = input.bool(true, "Show 2 Days Back High Line")
showMidLine    = input.bool(true, "Show 3 Days Back Mid Line")
highLineColor  = input.color(color.blue, "2 Days Back High Color")
midLineColor   = input.color(color.orange, "3 Days Back Mid Color")
lineWidth      = input.int(1, "Line Width", minval=1, maxval=5)

// Detect new day
newDay = timeframe.change("D")

// Current day tracking
var float curDayHigh = na
var float curDayLow  = na

// Completed daily values
var float[] dayHighArr = array.new_float()
var float[] dayLowArr  = array.new_float()

if barstate.isfirst
    curDayHigh := high
    curDayLow := low

if newDay
    array.push(dayHighArr, curDayHigh)
    array.push(dayLowArr, curDayLow)

    curDayHigh := high
    curDayLow := low
else
    curDayHigh := na(curDayHigh) ? high : math.max(curDayHigh, high)
    curDayLow  := na(curDayLow)  ? low  : math.min(curDayLow, low)

// Levels
twoDaysBackHigh = array.size(dayHighArr) >= 2 ? array.get(dayHighArr, array.size(dayHighArr) - 2) : na

threeDaysBackLow  = array.size(dayLowArr)  >= 3 ? array.get(dayLowArr,  array.size(dayLowArr)  - 3) : na
threeDaysBackMid  = not na(threeDaysBackHigh) and not na(threeDaysBackLow) ? (threeDaysBackHigh + threeDaysBackLow) / 2.0 : na

// Store the bar where level becomes available/calculated
var int twoDaysBackCalcBar = na
var int threeDaysBackCalcBar = na

if newDay
    if not na(twoDaysBackHigh)

    if not na(threeDaysBackMid)
        threeDaysBackCalcBar := bar_index-3
